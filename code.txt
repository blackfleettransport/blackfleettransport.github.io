// Black Fleet Transport: يحفظ كل تسجيل من الموقع بجدول "التسجيلات" ويرسل إيميل تنبيه.
// كل اللي تسويه: Deploy ← New deployment ← Web app ← Deploy ← انسخ الرابط.

const SHEET_ID = '18OhtccaMMawuOHbpXZm_2UtqxIQtXZJE7iorIClxQFM';
const NOTIFY_EMAIL = 'blackfleet.transport@gmail.com';  // يوصله تنبيه بكل تسجيل جديد. امسحه إذا ما تريد تنبيه.

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const d = JSON.parse(e.postData.contents || '{}');
    const clip = function (v) { return String(v == null ? '' : v).slice(0, 500); };
    // حماية من الإدخال الخبيث بالجدول: نص يبدي بـ = + - @ ينكتب كنص مو معادلة
    const safe = function (v) { v = clip(v); return /^[=+\-@]/.test(v) ? "'" + v : v; };
    if (!clip(d.name).trim() || !clip(d.phone).trim()) return out({ ok: false });
    if (d.website) return out({ ok: true });  // فخ للروبوتات: حقل مخفي، الإنسان ما يعبيه

    const sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName('التسجيلات');
    sheet.appendRow([
      new Date(), safe(d.name), safe(d.phone), safe(d.country), safe(d.vehicle),
      safe(d.licence), safe(d.experience), safe(d.notes), safe(d.lang), false
    ]);
    sheet.getRange(sheet.getLastRow(), 10).insertCheckboxes();

    if (NOTIFY_EMAIL) {
      MailApp.sendEmail(NOTIFY_EMAIL, 'سائق جديد سجّل: ' + clip(d.name),
        'الاسم: ' + clip(d.name) + '\nالهاتف: ' + clip(d.phone) + '\nالبلد: ' + clip(d.country) +
        '\nالسيارة: ' + clip(d.vehicle) + '\nالإجازة: ' + clip(d.licence) + '\nالخبرة: ' + clip(d.experience) +
        '\nملاحظات: ' + clip(d.notes) + '\n\n' + SpreadsheetApp.openById(SHEET_ID).getUrl());
    }
    return out({ ok: true });
  } finally {
    lock.releaseLock();
  }
}

function out(o) {
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}

// للتجربة اليدوية: شغّل هذي الدالة مرة وحدة حتى توافق على الصلاحيات
function authorize() {
  SpreadsheetApp.openById(SHEET_ID).getName();
  MailApp.getRemainingDailyQuota();
}
